# program to check leap year

year = int(input('Enter a year: '))

if year % 4 == 0 and year % 100 != 0 or year % 400 == 0:
    print('Leap Year')
else:
    print('Not a leap year')

Output: Enter a year: 1900
        Not a leap year
        Enter a year: 2008
        Leap Year

# Program to check vowel or consonant

char1 = input('Enter a character: ').lower()

if len(char1) == 1 and char1.isalpha():
    if char1 in 'aeiou':
        print('The character is vowel')
    else:
        print('The character is consonant')

else:
    print('Invalid character')

Output: Enter a character: u
        The character is vowel
        Enter a character: g
        The character is consonant

# Get grade input from the user

grade = input("Enter your grade (A, B, C, D, F): ").upper()  

match grade:
    case "A":
        print("Excellent!")
    case "B":
        print("Very Good!")
    case "C":
        print("Good")
    case "D":
        print("Needs Improvement")
    case "F":
        print("Fail")
    case _:  
        print("Invalid grade entered")

Output: Enter your grade (A, B, C, D, F): A
             Excellent!

# Get traffic light color input from the user

color = input("Enter traffic light color: ").capitalize()  # Capitalize to standardize input

match color:
    case "Red":
        print("Stop")
    case "Yellow":
        print("Wait")
    case "Green":
        print("Go")
    case _:
        print("Invalid color entered")

Output: Enter traffic light color: green
             Go

#Program to print factorial of a number

num = int(input("Enter a number: "))

factorial = 1

for i in range(1, num + 1):
    factorial *= i
print('The factorial of',num , 'is', factorial)

Output: Enter a number: 5
             The factorial of 5 is 120

# Program to print numbers from 1 to 10, but skips printing the number 5 (use continue)
# and stops the loop if the number 8 is reached (use break).

for i in range(1, 11):
    if i == 5:
        continue
    elif i == 8:
        break
    print(i, end='\t')

Output: 1	2	3	4	6	7
